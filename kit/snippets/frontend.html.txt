--- 3. WebAPP/App/Controller/DataFile.js

   a) read the dropdown instead of the hardcoded solver:

      OLD:  let solver = 'cbc';
      NEW:  let solver = $('#osy-solverSelect').val() || 'cbc';

   b) show/hide the dropdown together with the RUN button - append to every
      line that shows or hides #osy-run:

      $("#osy-run").show();  ->  $("#osy-run").show(); $("#osy-solverSelect").show();
      $("#osy-run").hide();  ->  $("#osy-run").hide(); $("#osy-solverSelect").hide();

   c) fill the HiGHS tab - add after each place that fills #osy-lpOutput
      (there are usually three: success, warning, error):

      $("#osy-highsOutput").empty();
      if (response.highs_message) {
          $(".highsOutput").show();
          $("#osy-highsOutput").html('<pre class="log-output">' + response.highs_message + '</pre>');
      } else {
          $(".highsOutput").hide();
      }

   d) hide the tab together with the other log tabs:

      $(".lpOutput").hide();  ->  $(".lpOutput").hide(); $(".highsOutput").hide();
